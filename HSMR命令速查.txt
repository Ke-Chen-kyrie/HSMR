# 6.1 触发渲染 POST /joint (8011 render; 检测到人才成功, 抓 foxglove 相机流)
#     选中左手+头 (value=1 表示选中):
curl -s -X POST http://127.0.0.1:8011/joint \
  -H "Content-Type: application/json" \
  -d '{"hand_l":1,"head":1}'

# 全身整体渲染 (body 关键字 = 全部 24 关节):
curl -s -X POST http://127.0.0.1:8011/joint \
  -H "Content-Type: application/json" \
  -d '{"body":1}'

# 只渲染头, 指定渲染第 2 个人 (person_index=1, 0 起):
curl -s -X POST http://127.0.0.1:8011/joint?person_index=1 \
  -H "Content-Type: application/json" \
  -d '{"head":1}'

# 一次渲染所有检测到的人 (person_index=all):
curl -s -X POST "http://127.0.0.1:8011/joint?person_index=all" \
  -H "Content-Type: application/json" \
  -d '{"head":1,"hand_r":1}'

# 响应示例:
# {"queued":true,"queue_size":1,"timestamp":1788320014.78,"rendered_seq":0,
#  "next_frames":4,"person_index":0,"all_persons":false,"num_persons":2,
#  "joints":["head"],"joints_cn":["头"],"glow":true,"infer_ms":1875.7,"bg_started":true}

# 6.2 上传图片推理 POST /infer (8010 纯推理 / 8011 渲染, 同一份输入):
#     只传彩色图:
curl -s -X POST http://127.0.0.1:8010/infer \
  -F "image=@/path/to/rgb.jpg"

#     彩色 + 深度融合 (需同时给 fx/fy/cx/cy):
curl -s -X POST http://127.0.0.1:8010/infer \
  -F "image=@/path/to/rgb.jpg" \
  -F "depth=@/path/to/depth.png" \
  -F "fx=600" -F "fy=600" -F "cx=640" -F "cy=360"

#     render 同款 (接收 8010 转发的同一份输入):
curl -s -X POST http://127.0.0.1:8011/infer \
  -F "image=@/path/to/rgb.jpg"

# 6.3 取渲染结果 (队列状态 / 取一张图, 取出即移除):
curl -s http://127.0.0.1:8011/render/queue
curl -s http://127.0.0.1:8011/render/pop | python3 -c "
import sys, json, base64
d = json.load(sys.stdin)
r = d.get('rendered')
if r:
    open('/tmp/render.jpg','wb').write(base64.b64decode(r['image_base64']))
    print('saved', r['joints'], '-> /tmp/render.jpg')
else:
    print('queue empty')
"

# 一条命令: 找 proactive_agent 的 PID + 跟踪它在读写什么
PID=$(pgrep -f "proactive_agent.*main.py") && echo "PID=$PID" && strace -p $PID -f -tt -e trace=read,write 2>&1 | tail -200

# 两步版: 找 PID + 实时看它的标准输出 (无需 sudo)
PID=$(pgrep -f "proactive_agent.*main.py") && echo "PID=$PID" && cat /proc/$PID/fd/1 && tail -f /proc/$PID/fd/1

# 查所有 agent 进程 / 容器

docker ps -a --format "table {{.Names}}\t{{.Status}}\t{{.Image}}" | grep -i agent
